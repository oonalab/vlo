    Function defineWorker

    • defineWorker(impls: ActivityImpls, options?: WorkerOptions): Worker

      Defines a worker: a map from activity name to ActivityHandler plus the queues it serves. Passed to createHost({ workers }) for in-process execution, runPullWorker for long-polling, or found by the config loader's workers globs.

      The worker holds no schemas; validation happens in the reducer (input) and on the host (output). Throws a TypeError when impls is empty.

      Parameters

      • impls: ActivityImpls
      • options: WorkerOptions = {}

      Returns Worker

      Example

      import { defineWorker } from "@vlo/sdk";

      export default defineWorker(
        {
          "example.plan": (input: { n: number }) => ({
            items: Array.from({ length: input.n }, (_, i) => `item-${i}`),
          }),
          "example.work": async (input: { item: string }, ctx) => {
            await ctx.heartbeat();
            ctx.reportUsage({ tokens: 12 });
            return { upper: input.item.toUpperCase() };
          },
        },
        { queues: ["default"] },
      );
