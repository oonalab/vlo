1//! Envelope construction shared by `prepare` and `reduce`.
2//!
3//! Lifecycle events (`task/scheduled`, `timer/scheduled`, …) inherit `ts_ms` and
4//! `random` from the trigger event: they are consequences of the same transition, and
5//! re-using the trigger's values keeps a replay byte-identical.
6
7use vlo_types::{Envelope, EventInput, Kind, RunRef, Status};
8
9/// The default user state a lifecycle transition echoes when the run has no checkpoint
10/// yet (only reachable before `started` has been folded).
11pub const EMPTY_STATE: &str = "null";
12
13/// The parts of a lifecycle event that differ from the trigger.
14pub struct LifecycleEvent<'a> {
15    pub step_id: &'a str,
16    pub attempt: u32,
17    pub kind: Kind,
18    pub status: Status,
19    pub schema_ref: Option<String>,
20    pub payload: Option<String>,
21}
22
23/// Build the `EventInput` of a lifecycle event from the trigger's `ts_ms` / `random`.
24pub fn lifecycle_input(
25    run_id: &str,
26    ts_ms: u64,
27    random: &str,
28    event: LifecycleEvent<'_>,
29) -> EventInput {
30    EventInput {
31        run_id: run_id.to_string(),
32        step_id: event.step_id.to_string(),
33        attempt: event.attempt,
34        kind: event.kind,
35        status: event.status,
36        cause: None,
37        usage: None,
38        ts_ms,
39        random: random.to_string(),
40        schema_ref: event.schema_ref,
41        payload: event.payload,
42        payload_ref: None,
43        error: None,
44    }
45}
46
47/// Seal an `EventInput` into a journaled `Envelope`. `state_hash` is stamped later,
48/// once the whole transition is known, because every event in one transition carries
49/// the hash of the state that transition produced.
50pub fn envelope(input: EventInput, seq: u64, run: &RunRef) -> Envelope {
51    Envelope {
52        input,
53        seq,
54        parent_run_id: run.parent_run_id.clone(),
55        parent_step_id: run.parent_step_id.clone(),
56        state_hash: String::new(),
57    }
58}
