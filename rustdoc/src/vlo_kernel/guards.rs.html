1//! Shared preconditions for `prepare_event`: the checks that decide whether an event
2//! may touch a step at all, before anything is folded.
3
4use serde_json::Value;
5use vlo_types::{EventInput, KernelState, Kind, PrepareEventInput, Status, StepState};
6
7use crate::canonical;
8use crate::error::{code, err, err_step, KResult};
9use crate::step_machine::is_legal;

11/// The kernel state, or `invalid_event` when absent (only `started` may omit it).
12pub fn require_kernel_state(input: &PrepareEventInput) -> KResult<&KernelState> {
13    input.kernel_state.as_ref().ok_or_else(|| {
14        err(
15            code::INVALID_EVENT,
16            "kernel_state is required for every event except started",
17        )
18    })
19}
20
21/// The open step the event names, or `unknown_step`; `illegal_transition` when its kind
22/// is not `kind`.
23pub fn require_step<'a>(
24    kernel_state: &'a KernelState,
25    event: &EventInput,
26    kind: Kind,
27) -> KResult<&'a StepState> {
28    let step = kernel_state.steps.get(&event.step_id).ok_or_else(|| {
29        err_step(
30            code::UNKNOWN_STEP,
31            "no open step with this id",
32            &event.step_id,
33        )
34    })?;
35    if step.kind != kind {
36        return Err(err_step(
37            code::ILLEGAL_TRANSITION,
38            format!("step is a {:?}, event is a {kind:?}", step.kind),
39            &event.step_id,
40        ));
41    }
42    Ok(step)
43}

45/// Reject an event whose `attempt` is not the step's current one: `already_terminal` for
46/// a past attempt, `illegal_transition` for a future one.
47pub fn check_attempt(step: &StepState, event: &EventInput) -> KResult<()> {
48    if event.attempt == step.attempt {
49        return Ok(());
50    }
51    let code = if event.attempt < step.attempt {
52        code::ALREADY_TERMINAL
53    } else {
54        code::ILLEGAL_TRANSITION
55    };
56    Err(err_step(
57        code,
58        format!(
59            "event is for attempt {}, step is on attempt {}",
60            event.attempt, step.attempt
61        ),
62        &event.step_id,
63    ))
64}

66/// Reject the edge `step.status → event.status` when the state machine has no such edge
67/// (`illegal_transition`).
68pub fn check_legal(step: &StepState, event: &EventInput) -> KResult<()> {
69    if is_legal(step.status, event.status) {
70        return Ok(());
71    }
72    Err(err_step(
73        code::ILLEGAL_TRANSITION,
74        format!(
75            "{:?} → {:?} is not a legal step edge",
76            step.status, event.status
77        ),
78        &event.step_id,
79    ))
80}

82/// Set the status of an open step; a no-op for an unknown id.
83pub fn set_status(kernel_state: &mut KernelState, step_id: &str, status: Status) {
84    if let Some(step) = kernel_state.steps.get_mut(step_id) {
85        step.status = status;
86    }
87}

89/// The `delay_ms` (milliseconds) a `retrying` event carries in its payload object; `0`
90/// when there is no payload or no `delay_ms`. Errors with `invalid_event` on any other shape.
91pub fn retry_delay_ms(event: &EventInput) -> KResult<u64> {
92    let Some(payload) = &event.payload else {
93        return Ok(0);
94    };
95    match canonical::parse(payload)? {
96        Value::Object(map) => match map.get("delay_ms") {
97            None => Ok(0),
98            Some(Value::Number(n)) => n.as_u64().ok_or_else(|| {
99                err_step(
100                    code::INVALID_EVENT,
101                    "delay_ms must be a non-negative integer",
102                    &event.step_id,
103                )
104            }),
105            Some(_) => Err(err_step(
106                code::INVALID_EVENT,
107                "delay_ms must be a number",
108                &event.step_id,
109            )),
110        },
111        _ => Err(err_step(
112            code::INVALID_EVENT,
113            "a retrying event payload must be an object",
114            &event.step_id,
115        )),
116    }
