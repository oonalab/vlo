1//! Lifecycle transitions: the write-set for events that need no reducer call.
2//!
3//! They carry the trigger event, the updated kernel state, and whatever task updates
4//! the bookkeeping implies. The user state is echoed unchanged from the checkpoint the
5//! host passed in — but it is still hashed together with the new kernel state, because
6//! the kernel state moved.
7
8use vlo_types::INLINE_LIMIT_BYTES;
9use vlo_types::{EventInput, KernelState, RunPatch, RunRef, RunStatus, TaskUpdate, Transition};
10
11use crate::error::KResult;
12use crate::events;
13use crate::hash;
14
15/// Everything `build` needs besides the trigger event.
16pub struct LifecycleTransition<'a> {
17    pub run: &'a RunRef,
18    pub kernel_state: KernelState,
19    pub state: &'a str,
20    pub run_status: RunStatus,
21    pub seq: u64,
22    pub task_updates: Vec<TaskUpdate>,
23}
24
25/// Assemble the lifecycle write-set: one event, the new kernel state, the echoed user state.
26pub fn build(event: &EventInput, spec: LifecycleTransition<'_>) -> KResult<Transition> {
27    let state_hash = hash::state_hash(&spec.kernel_state, spec.state)?;
28    let mut envelope = events::envelope(event.clone(), spec.seq, spec.run);
29    envelope.state_hash = state_hash.clone();
30
31    Ok(Transition {
32        run: RunPatch {
33            status: spec.run_status,
34            last_seq: spec.seq,
35        },
36        kernel_state: spec.kernel_state,
37        state: spec.state.to_string(),
38        state_hash,
39        state_inline: spec.state.len() <= INLINE_LIMIT_BYTES,
40        events: vec![envelope],
41        tasks: Vec::new(),
42        task_updates: spec.task_updates,
43        timers: Vec::new(),
44        spawns: Vec::new(),
45        signal_waits: Vec::new(),
46        emits: Vec::new(),
47        completion: None,
48        continue_as_new: None,
49    })
50}
