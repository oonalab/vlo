1//! Operations on `KernelState::steps`.
2//!
3//! Terminal steps are pruned rather than kept with a terminal status, so the map stays
4//! bounded on long runs; `closed_steps` carries the count. A later event for a pruned
5//! step therefore reports `unknown_step`, which is exactly the answer a late duplicate
6//! result deserves.
7
8use vlo_types::{KernelState, Kind, Status, StepState, TaskUpdate};
9
10use crate::classify::{timeout_owner, timeout_step_id};
11use crate::ids;
12
13/// Prune one step and account for it.
14pub fn close_step(kernel_state: &mut KernelState, step_id: &str) -> Option<StepState> {
15    let step = kernel_state.steps.remove(step_id)?;
16    kernel_state.closed_steps = kernel_state.closed_steps.saturating_add(1);
17    Some(step)
18}
19
20/// Close every open step and emit a `cancelled` task update for each open task, so the
21/// host releases leases instead of leaving workers running for a dead run.
22pub fn close_open_steps(kernel_state: &mut KernelState, run_id: &str) -> Vec<TaskUpdate> {
23    let open: Vec<(String, StepState)> = kernel_state
24        .steps
25        .iter()
26        .map(|(id, step)| (id.clone(), step.clone()))
27        .collect();
28
29    let mut updates = Vec::new();
30    for (step_id, step) in &open {
31        if step.kind == Kind::Task {
32            updates.push(TaskUpdate {
33                task_id: ids::task_id(run_id, step_id),
34                attempt: step.attempt,
35                status: Status::Cancelled,
36                not_before_ms: None,
37            });
38        }
39    }
40    kernel_state.closed_steps = kernel_state.closed_steps.saturating_add(open.len() as u64);
41    kernel_state.steps.clear();
42    updates
43}

45/// Whether the timer `timer_id` is the `<id>:timeout` deadline of a signal step that is
46/// still awaiting its signal.
47fn is_signal_deadline(kernel_state: &KernelState, timer_id: &str) -> bool {
48    timeout_owner(timer_id).is_some_and(|owner| {
49        kernel_state
50            .steps
51            .get(owner)
52            .is_some_and(|step| step.kind == Kind::Signal && step.status == Status::AwaitingSignal)
53    })
54}

56/// When a signal step or its `<id>:timeout` deadline timer settles, close the other side
57/// of the race as `cancelled` so the reducer only ever sees the winner. Timers are not
58/// tasks, so no task update is owed; the loser is pruned like any other terminal step,
59/// and a late event for it reports `unknown_step`.
60pub fn cancel_race_loser(kernel_state: &mut KernelState, kind: Kind, step_id: &str) {
61    let loser = match kind {
62        Kind::Signal => {
63            let timer_id = timeout_step_id(step_id);
64            let scheduled = kernel_state
65                .steps
66                .get(&timer_id)
67                .is_some_and(|step| step.kind == Kind::Timer && step.status == Status::Scheduled);
68            scheduled.then_some(timer_id)
69        }
70        Kind::Timer if is_signal_deadline(kernel_state, step_id) => {
71            timeout_owner(step_id).map(str::to_string)
72        }
73        _ => None,
74    };
75    if let Some(loser) = loser {
76        close_step(kernel_state, &loser);
77    }
78}

80/// Whether any open step can still produce work (task / timer / child). A deadline timer
81/// paired with an open signal step belongs to that wait, so it does not count: the run is
82/// parked on the signal, deadline or not.
83pub fn has_open_work(kernel_state: &KernelState) -> bool {
84    kernel_state.steps.iter().any(|(id, step)| match step.kind {
85        Kind::Task | Kind::Child => true,
86        Kind::Timer => !is_signal_deadline(kernel_state, id),
87        Kind::Signal | Kind::Marker => false,
88    })
89}

91/// Whether the run is parked on at least one signal.
92pub fn has_open_signal(kernel_state: &KernelState) -> bool {
93    kernel_state
94        .steps
95        .values()
96        .any(|step| step.kind == Kind::Signal)
