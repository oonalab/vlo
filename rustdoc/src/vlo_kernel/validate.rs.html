1//! Envelope validation. Runs before every other export so no later stage has to
2//! re-check field shape.
3
4use serde_json::Value;
5use vlo_types::{EventInput, INLINE_LIMIT_BYTES};
6
7use crate::canonical;
8use crate::error::{code, err, err_step, KResult};

10/// `random` is 32 lowercase hex chars of journaled entropy.
11pub const RANDOM_HEX_LEN: usize = 32;
12
13/// Prefix of an offloaded payload reference.
14pub const BLOB_REF_PREFIX: &str = "blob://sha256/";
15
16/// Whether `value` is exactly [`RANDOM_HEX_LEN`] lowercase hex chars.
17pub fn is_random_hex(value: &str) -> bool {
18    value.len() == RANDOM_HEX_LEN
19        && value
20            .bytes()
21            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
22}

24/// Field/enum checks plus the inline payload limit. Returns the normalized event; the
25/// kernel keeps payloads byte-exact, so normalization is currently a no-op clone that
26/// exists so the host has one place to adopt future coercions.
27pub fn validate_event(event: &EventInput) -> KResult<EventInput> {
28    if event.run_id.is_empty() {
29        return Err(err(code::INVALID_EVENT, "run_id is empty"));
30    }
31    if event.step_id.is_empty() {
32        return Err(err(code::INVALID_EVENT, "step_id is empty"));
33    }
34    if !is_random_hex(&event.random) {
35        return Err(err_step(
36            code::INVALID_EVENT,
37            "random must be 32 lowercase hex chars",
38            &event.step_id,
39        ));
40    }
41    if event.payload.is_some() && event.payload_ref.is_some() {
42        return Err(err_step(
43            code::INVALID_EVENT,
44            "at most one of payload / payload_ref",
45            &event.step_id,
46        ));
47    }
48    if let Some(reference) = &event.payload_ref {
49        if !reference.starts_with(BLOB_REF_PREFIX) {
50            return Err(err_step(
51                code::INVALID_EVENT,
52                format!("payload_ref must start with {BLOB_REF_PREFIX}"),
53                &event.step_id,
54            ));
55        }
56    }
57    if let Some(payload) = &event.payload {
58        if payload.len() > INLINE_LIMIT_BYTES {
59            return Err(err_step(
60                code::PAYLOAD_TOO_LARGE,
61                format!(
62                    "inline payload is {} bytes, limit is {INLINE_LIMIT_BYTES}",
63                    payload.len()
64                ),
65                &event.step_id,
66            ));
67        }
68        canonical::parse(payload)?;
69    }
70    Ok(event.clone())
71}

73/// The payload as a value. `payload_ref` must be resolved by the host before the kernel
74/// sees it, because a reducer event carries the value, not the pointer.
75pub fn payload_value(event: &EventInput) -> KResult<Value> {
76    if event.payload_ref.is_some() {
77        return Err(err_step(
78            code::INVALID_EVENT,
79            "payload_ref must be resolved to an inline payload before folding",
80            &event.step_id,
81        ));
82    }
83    match &event.payload {
84        Some(payload) => canonical::parse(payload),
85        None => Ok(Value::Null),
86    }
87}
88
89/// Like [`payload_value`] but distinguishes "absent" from "present and null".
90pub fn optional_payload_value(event: &EventInput) -> KResult<Option<Value>> {
91    if event.payload.is_none() && event.payload_ref.is_none() {
92        return Ok(None);
93    }
94    payload_value(event).map(Some)
95}
