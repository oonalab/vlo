1//! Adapter traits for a Rust host (see `docs/journal.md`). Not yet used by any host —
2//! the TypeScript host implements the same interfaces natively — but kept in sync so a
3//! Rust host can reuse the same write-set contract.
4
5#![deny(missing_docs)]
6
7use vlo_types::{Envelope, KernelState, Transition};
8
9/// The run's checkpoint: kernel state and user state after the transition at `seq`.
10pub struct StateRow {
11    /// Seq of the transition that wrote this row; always the run's `last_seq`.
12    pub seq: u64,
13    /// Bundle the state was produced by. A mismatch with the run's pin forces a fold from seq 0.
14    pub bundle_digest: String,
15    /// Kernel-owned bookkeeping.
16    pub kernel_state: KernelState,
17    /// User state, byte-exact JSON.
18    pub state: String,
19    /// Hash of `(kernel_state, state)`; verified on every load.
20    pub state_hash: String,
21}
22
23/// Ordered, per-run journal. `apply_transition` is the only write path for history.
24pub trait Journal {
25    /// Adapter-specific failure.
26    type Error;
27    /// Events with `seq > since_seq`, ascending.
28    fn read(&self, run_id: &str, since_seq: u64) -> Result<Vec<Envelope>, Self::Error>;
29    /// The latest checkpoint, or `None` before the first transition was applied.
30    fn get_state(&self, run_id: &str) -> Result<Option<StateRow>, Self::Error>;
31    /// Apply one write-set atomically under the run lock.
32    fn apply_transition(&self, run_id: &str, t: &Transition) -> Result<(), Self::Error>;
33}
34
35/// Content-addressed storage for payloads and state above the inline limit
36/// (see `docs/limits.md`).
37pub trait BlobStore {
38    /// Adapter-specific failure.
39    type Error;
40    /// Store `bytes`; returns the lowercase hex sha256 that names them.
41    fn put(&self, bytes: &[u8]) -> Result<String, Self::Error>;
42    /// Fetch the bytes named by `sha256_hex`.
43    fn get(&self, sha256_hex: &str) -> Result<Vec<u8>, Self::Error>;
44}
45
46/// Delivers tasks to workers. HTTP push is the reference implementation.
47pub trait TaskTransport {
48    /// Adapter-specific failure.
49    type Error;
50    /// Deliver one task; `Ok(true)` when the worker accepted it (202).
51    fn push(&self, queue: &str, task_json: &str) -> Result<bool, Self::Error>;
52}
